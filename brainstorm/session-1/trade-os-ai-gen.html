// V1 · 关键词路由 · 0 依赖

用户输入: "做一个装箱单" → 关键词匹配 → 打开 packing-list.html
var AI_ROUTES = { '发票': 'commercial-invoice.html', '装箱单': 'packing-list.html', '提单': 'bill-of-lading.html', '合同': 'trade-contract.html', '报价': 'commercial-quotation.html', '原产地': 'certificate-of-origin.html', '保险': 'insurance-policy.html', '报关': 'customs-declaration.html', '熏蒸': 'fumigation-certificate.html', '成本': 'cost-estimation.html', '检验': 'inspection-certificate.html', '发货人委托': 'shipper-letter-instruction.html', // ... 英文版 + 更多同义词 }; function handleAiCommand(input) { var q = input.toLowerCase(); for (var key in AI_ROUTES) { if (q.indexOf(key) !== -1) { window.location.href = AI_ROUTES[key]; return; } } // 无匹配: 显示提示 showAiHint('试试: 发票 / 提单 / 装箱单 / 合同 ...'); }
v1 优劣: 零外部依赖，100% 纯前端，毫秒级响应。但只能做路由跳转，不能预填任何字段。
适用: 作为 AI 命令栏的初始体验，覆盖 90% 的用户意图（"我要做X" → 跳转到 X 页面）。

// V2 · LLM 智能填表 · 核心功能

用户输入: 自然语言描述 → LLM API → 结构化 JSON → 写入 data → 跳转页面
核心思路: 在首页 AI 命令栏输入自然语言 → 调用 LLM API 提取字段 → 把结构化数据写入 localStorage → 跳转到对应页面 → 页面加载时读取预填数据 → 用户确认/修改。
系统 Prompt 设计（关键）: 告诉 LLM 分析用户输入，判断文档类型，提取所有字段值，返回 JSON。
var AI_SYSTEM_PROMPT = `你是 Trade OS 的 AI 助手。分析用户输入，返回 JSON: { "docType": "commercial_invoice|bill_of_lading|packing_list|...", "confidence": 0.95, "fields": { "shipperCompany": "...", "shipperAddress": "...", "consigneeCompany": "...", "invoiceNo": "...", "currency": "USD", "items": [ { "description": "...", "qty": "100", "unitValue": "8.50" } ] }, "suggestion": "已为您生成商业发票，请检查后确认" } 规则: - docType 必须是以下之一: commercial_invoice, packing_list, bill_of_lading, trade_contract, ... - 如果用户没提字段，就不要猜，留空字符串 "" - currency 默认 USD，金额默认 2 位小数 - 中文输入优先识别中文公司名和地址`;
完整流程:
1. 用户输入 → 2. 显示 loading 动画 → 3. fetch LLM API → 4. 解析 JSON → 5. 写入 localStorage → 6. 跳转页面
function handleAiGenerate(input) { var bar = document.getElementById('ai-bar'); bar.classList.add('loading'); callLLM(input).then(function(result) { bar.classList.remove('loading'); if (!result.docType || !result.fields) { showAiHint('未能理解您的需求，请尝试更具体的描述'); return; } // 把 AI 提取的字段预存到 localStorage var aiData = { fields: result.fields, suggestion: result.suggestion || '', timestamp: Date.now() }; localStorage.setItem(userKey('ai_prefill_' + result.docType), JSON.stringify(aiData)); // 跳转到对应页面 var page = DOC_META[result.docType].page; window.location.href = page; }); }

// LLM API 接入方案

推荐方案: 用户自带 API Key → 存 localStorage → fetch OpenAI 兼容接口
  • API 配置界面: 在首页或全局导航添加"设置"入口 → 简单的 API Key 输入框 + 端点 URL（默认 OpenAI） + 模型选择
  • 隐私保证: API Key 只存 localStorage，每次请求由浏览器直接发送，不经过任何中间服务器
  • 降级策略: 如果用户没配 API Key → 自动降级到 v1 关键词路由；API 调用失败 → 降级到 v1
  • 支持模型: 任何 OpenAI 兼容接口（GPT-4o, GPT-4o-mini, DeepSeek, 智谱 GLM 等）
  • 费用: 用户自付 API 费用。用 GPT-4o-mini 做单证字段提取，每次约 $0.001（几乎免费）
/** 调用 LLM API */ function callLLM(userInput) { var config = getAiConfig(); // 从 localStorage 读 API Key + endpoint if (!config.apiKey) return v1Fallback(userInput); return fetch(config.endpoint + '/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + config.apiKey }, body: JSON.stringify({ model: config.model || 'gpt-4o-mini', messages: [ { role: 'system', content: AI_SYSTEM_PROMPT }, { role: 'user', content: userInput } ], response_format: { type: 'json_object' }, temperature: 0.1 }) }).then(function(r) { return r.json(); }) .then(function(d) { return JSON.parse(d.choices[0].message.content); }); }

// 目标页面 · 接收 AI 预填数据

在每个单证页面的 init() 中，加载数据前先检查是否有 AI 预填数据：
/* 在 init() 中 loadDocData 之前插入 */ function init() { applyTheme(); // ... 模板数据初始化 ... var saved = loadDocData(DT); var aiPrefill = loadAiPrefill(DT); // 检查 AI 预填 if (aiPrefill && !saved) { // 首次使用：合并 AI 预填到默认数据 var defaults = getDefaultDoc(DT); data = mergeAiFields(defaults, aiPrefill.fields); saveDocData(DT, data); // 清除预填标记（避免下次再覆盖） localStorage.removeItem(userKey('ai_prefill_' + DT)); // 显示 AI 提示 toast showAiSuggestion(aiPrefill.suggestion); } else if (saved) { data = saved; } else { data = defaults; } // ... }
关键判断: 只当用户没有已有数据（saved 为空）时才应用 AI 预填。已有表单数据的页面不会被覆盖。
AI Toast: 页面顶部显示一条琥珀色提示条："AI 已为您生成商业发票，请检查并修改。"
